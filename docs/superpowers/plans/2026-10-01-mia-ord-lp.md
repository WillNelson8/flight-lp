# MIA to ORD Same-Day LP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scrape real Nov 20, 2026 nonstop prices between 22 airports, solve the cheapest same-day MIA to ORD itinerary as a time-space network LP (plus K-best via cuts) for four scenarios, and present it in a Streamlit dashboard with a ranked route map.

**Architecture:**
- `fetch.py` writes one run directory of CSV and JSON files.
- `network.py` turns flights into a time-space network.
- `model.py` builds and solves the PuLP model on that network and cross-checks it with networkx.
- `charts.py` builds the Plotly figures, and `dashboard.py` lays them out. Neither one scrapes.

**Tech Stack:**
- Python 3.11 in `.venv`
- fast-flights 3.1.0 (request building and fetching only; our own payload parser)
- PuLP 3.3 with CBC
- networkx 3.6
- Streamlit 1.64 and Plotly 7.1
- pandas 3.0
- pytest 9.1

**Spec:** `docs/superpowers/specs/2026-10-01-mia-ord-lp-design.md`

## Global Constraints

- Trip date `2026-11-20`; origin `MIA`; destination `ORD`; 20 hubs exactly as listed in spec section 4.
- 421 nonstop searches plus 2 single-ticket reference searches; 1 adult, economy, one-way, `language="en-US"`, `currency="USD"`.
- Random 2 to 4 second pause between requests; stop after 5 consecutive failures; never bypass a CAPTCHA.
- `is_american` means marketing carrier code `"AA"`.
- Default MCT 45 minutes (slider 30 to 120, step 15); default K 5 (1 to 10).
- Deadline: ORD arrival no later than `2026-11-20 23:59 America/Chicago`.
- Tie term: `0.00003` per flight plus `0.5 * minutes_after_midnight_CT / 1440` for flights into ORD. Reported costs exclude it.
- No fabricated numbers anywhere: missing data displays as missing.
- UI copy: no emojis, no em dashes; light theme, single blue accent, gray context, Inter font, tabular figures.

## Review Focus

1. Google serves a block or CAPTCHA page partway through: the fetch stops after 5 in a row, keeps progress, and a re-run resumes without duplicating flights. (Task 3 tests this with a fake fetcher.)
2. A flight that lands after local midnight carries the arrival date from the payload, not the departure date. (Task 2: the 22:07 MIA-ORD flight lands 2026-11-21.)
3. A flight landing at exactly 23:59 CT is allowed, and one landing at 00:05 is not. (Task 5 boundary test.)
4. A flight appearing in both Google's "best" and "other" lists is kept once. (Task 2 dedupe test.)
5. A scenario with no feasible itinerary, or a dashboard opened before any complete run exists, shows a message and does not crash. (Task 5 infeasible test; Task 8 browser check with an empty `data/`.)

```python
"""On-disk format of one scrape run.

data/<stamp>/flights.csv    one row per nonstop flight
data/<stamp>/routes.csv     one row per search attempt (last row per route wins)
data/<stamp>/drops.csv      validation drop counts per route and reason
data/<stamp>/reference.json Google's cheapest single-ticket MIA-ORD fares
data/<stamp>/COMPLETE       present once every route is ok or no_service
"""
import csv
import os
from pathlib import Path

DATA_DIR = Path(os.environ.get("FLIGHT_LP_DATA", Path(__file__).parent / "data"))
FLIGHT_COLUMNS = ["origin", "dest", "carrier", "airline", "flight_number", "dep_local", "arr_local",
                  "price_usd", "is_american", "aircraft", "scraped_at"]
ROUTE_COLUMNS = ["origin", "dest", "status", "n_flights", "scraped_at"]
DROP_COLUMNS = ["origin", "dest", "reason", "count"]


def new_run_dir(stamp: str, data_dir: Path = DATA_DIR) -> Path:
    run_dir = data_dir / stamp
    run_dir.mkdir(parents=True, exist_ok=False)
    return run_dir


def is_complete(run_dir: Path) -> bool:
    return (run_dir / "COMPLETE").exists()


def mark_complete(run_dir: Path) -> None:
    (run_dir / "COMPLETE").write_text("")


def latest_run(data_dir: Path = DATA_DIR, complete: bool = True) -> Path | None:
    """Most recent run directory that is complete (or, with complete=False, not yet complete)."""
    if not data_dir.exists():
        return None
    runs = sorted(p for p in data_dir.iterdir() if p.is_dir() and is_complete(p) == complete)
    return runs[-1] if runs else None


def append_rows(path: Path, columns: list[str], rows: list[dict]) -> None:
    if not rows:
        return
    new_file = not path.exists()
    with path.open("a", newline="") as fh:
        writer = csv.DictWriter(fh, fieldnames=columns)
        if new_file:
            writer.writeheader()
        writer.writerows(rows)


def read_rows(path: Path) -> list[dict]:
    if not path.exists():
        return []
    with path.open(newline="") as fh:
        return list(csv.DictReader(fh))


def route_status(run_dir: Path) -> dict[tuple[str, str], str]:
    status = {}
    for row in read_rows(run_dir / "routes.csv"):
        status[(row["origin"], row["dest"])] = row["status"]
    return status
```

```python
"""Scrape real Nov 20 nonstop prices from Google Flights for every route in the network.

Usage:
    python fetch.py          start a new run, or resume the latest unfinished one
    python fetch.py --new    force a new run
"""
import json
from collections import Counter
from datetime import datetime
from zoneinfo import ZoneInfo

from selectolax.lexbor import LexborHTMLParser

from airports import AIRPORTS, TRIP_DATE

TRIP_DAY = tuple(int(p) for p in TRIP_DATE.split("-"))


def parse_time(value) -> tuple[int, int]:
    """Google omits zero components: [8] is 08:00 and [None, 31] is 00:31."""
    padded = [*(value or []), None, None]
    return (padded[0] or 0, padded[1] or 0)


def parse_payload(html: str) -> list[dict] | None:
    """Itineraries from both of Google's result lists, or None if the page carries no payload."""
    node = LexborHTMLParser(html).css_first(r"script.ds\:1")
    if node is None or "data:" not in node.text():
        return None
    data = node.text().split("data:", 1)[1].rsplit(",", 1)[0]
    if data.endswith("errorHasStatus: true"):
        return []
    try:
        payload = json.loads(data)
    except json.JSONDecodeError:
        return None
    itineraries = []
    for section in (payload[2], payload[3]):  # Google's "best" and "other" lists
        if not section or not section[0]:
            continue
        for item in section[0]:
            legs = []
            for leg in item[0][2]:
                code = leg[22] or [None, None, None, None]
                legs.append({
                    "origin": leg[3], "dest": leg[6],
                    "carrier": code[0], "number": code[1], "airline": code[3],
                    "dep_date": tuple(leg[20]), "dep_time": parse_time(leg[8]),
                    "arr_date": tuple(leg[21]), "arr_time": parse_time(leg[10]),
                    "aircraft": leg[17],
                })
            price = item[1][0][1] if item[1] and item[1][0] else None
            itineraries.append({"price": price, "legs": legs})
    return itineraries


def _local(airport: str, date: tuple, hm: tuple) -> datetime:
    return datetime(*date, *hm, tzinfo=ZoneInfo(AIRPORTS[airport].tz))


def nonstop_rows(itineraries: list[dict], origin: str, dest: str, scraped_at: str) -> tuple[list[dict], Counter]:
    """Validated snapshot rows for one searched route, plus drop counts by reason."""
    rows, drops, seen = [], Counter(), set()
    for it in itineraries:
        if len(it["legs"]) != 1:
            drops["not_nonstop"] += 1
            continue
        leg = it["legs"][0]
        if (leg["origin"], leg["dest"]) != (origin, dest):
            drops["route_mismatch"] += 1
            continue
        if not it["price"] or it["price"] <= 0:
            drops["missing_price"] += 1
            continue
        if leg["dep_date"] != TRIP_DAY:
            drops["wrong_date"] += 1
            continue
        dep = _local(origin, leg["dep_date"], leg["dep_time"])
        arr = _local(dest, leg["arr_date"], leg["arr_time"])
        if arr <= dep:
            drops["arrival_not_after_departure"] += 1
            continue
        flight_number = f"{leg['carrier']} {leg['number']}"
        if (flight_number, dep) in seen:
            drops["duplicate"] += 1
            continue
        seen.add((flight_number, dep))
        rows.append({
            "origin": origin, "dest": dest, "carrier": leg["carrier"], "airline": leg["airline"],
            "flight_number": flight_number, "dep_local": dep.isoformat(), "arr_local": arr.isoformat(),
            "price_usd": it["price"], "is_american": leg["carrier"] == "AA", "aircraft": leg["aircraft"],
            "scraped_at": scraped_at,
        })
    return rows, drops


def cheapest_itinerary(itineraries: list[dict]) -> dict | None:
    """Google's cheapest single-ticket itinerary; ties go to the earlier final arrival."""
    priced = [it for it in itineraries if it["price"]]
    if not priced:
        return None

    def arrival_key(it):
        leg = it["legs"][-1]
        return (*leg["arr_date"], *leg["arr_time"])

    best = min(priced, key=lambda it: (it["price"], arrival_key(it)))
    return {
        "price": best["price"],
        "legs": [{
            "origin": l["origin"], "dest": l["dest"],
            "flight_number": f"{l['carrier']} {l['number']}", "airline": l["airline"],
            "dep_local": "{:04d}-{:02d}-{:02d}T{:02d}:{:02d}".format(*l["dep_date"], *l["dep_time"]),
            "arr_local": "{:04d}-{:02d}-{:02d}T{:02d}:{:02d}".format(*l["arr_date"], *l["arr_time"]),
        } for l in best["legs"]],
    }
```

```python
"""Time-space network: event nodes (airport, time) linked by flight arcs and zero-cost wait arcs."""
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo

from airports import AIRPORTS, DEST, ORIGIN, TRIP_DATE

TRIP_DAY = int(TRIP_DATE.split("-")[2])


@dataclass(frozen=True)
class Flight:
    id: str
    origin: str
    dest: str
    carrier: str
    airline: str
    flight_number: str
    dep: datetime  # timezone-aware, origin local
    arr: datetime  # timezone-aware, destination local
    price: int
    is_american: bool


def flight_id(flight_number: str, origin: str, dest: str, dep_local: datetime) -> str:
    return f"{flight_number.replace(' ', '')}_{origin}{dest}_{dep_local:%H%M}"


def flight_from_row(row: dict) -> Flight:
    dep = datetime.fromisoformat(row["dep_local"])
    return Flight(
        id=flight_id(row["flight_number"], row["origin"], row["dest"], dep),
        origin=row["origin"], dest=row["dest"], carrier=row["carrier"], airline=row["airline"],
        flight_number=row["flight_number"], dep=dep, arr=datetime.fromisoformat(row["arr_local"]),
        price=int(row["price_usd"]), is_american=str(row["is_american"]) == "True",
    )


@dataclass(frozen=True)
class Node:
    airport: str
    t: datetime  # UTC

    @property
    def local(self) -> datetime:
        return self.t.astimezone(ZoneInfo(AIRPORTS[self.airport].tz))

    @property
    def name(self) -> str:
        local = self.local
        suffix = "" if local.day == TRIP_DAY else f"_d{local.day:02d}"
        return f"{self.airport}_{local:%H%M}{suffix}"


@dataclass
class Network:
    flights: list[Flight]
    nodes: list[Node]
    waits: list[tuple[Node, Node]]
    tail: dict[str, Node | None]  # node a flight leaves from; None = leaves MIA
    head: dict[str, Node | None]  # node a flight feeds (arrival + MCT); None = lands at ORD
    mct: int


def build_network(flights: list[Flight], mct_min: int = 45) -> Network:
    usable = [f for f in flights if f.origin != DEST and f.dest != ORIGIN and f.origin != f.dest]
    tail, head, times = {}, {}, {}
    for f in usable:
        if f.origin == ORIGIN:
            tail[f.id] = None
        else:
            tail[f.id] = Node(f.origin, f.dep.astimezone(timezone.utc))
            times.setdefault(f.origin, set()).add(tail[f.id].t)
        if f.dest == DEST:
            head[f.id] = None
        else:
            head[f.id] = Node(f.dest, (f.arr + timedelta(minutes=mct_min)).astimezone(timezone.utc))
            times.setdefault(f.dest, set()).add(head[f.id].t)
    nodes, waits = [], []
    for airport in sorted(times):
        seq = [Node(airport, t) for t in sorted(times[airport])]
        nodes.extend(seq)
        waits.extend(zip(seq, seq[1:]))
    return Network(usable, nodes, waits, tail, head, mct_min)
```

```python
"""The LP: cheapest same-day MIA to ORD itinerary on the time-space network, plus K-best via no-good cuts.

Usage:
    python model.py [--k 5] [--mct 45] [--partial]
"""
import tempfile
from dataclasses import dataclass, field
from datetime import datetime
from pathlib import Path
from zoneinfo import ZoneInfo

import networkx as nx
import pulp

from airports import DEST, ORIGIN
from network import Flight, Network

CT = ZoneInfo("America/Chicago")
DEADLINE = datetime(2026, 11, 20, 23, 59, tzinfo=CT)
DAY_START = datetime(2026, 11, 20, 0, 0, tzinfo=CT)
TIE_PER_FLIGHT = 0.00003
TIE_ARRIVAL_WEIGHT = 0.5  # dollars spread across the 1440-minute day
GROUP_LABEL = {"C4": "sameday", "C5": "airline", "C6": "onestop"}


@dataclass(frozen=True)
class Scenario:
    key: str
    label: str
    american_only: bool
    one_stop: bool


SCENARIOS = [
    Scenario("S1", "American, any stops", True, False),
    Scenario("S2", "American, one stop at most", True, True),
    Scenario("S3", "All airlines, any stops", False, False),
    Scenario("S4", "All airlines, one stop at most", False, True),
]
SCENARIO_BY_KEY = {s.key: s for s in SCENARIOS}


def tie_term(f: Flight) -> float:
    """Far below $1: orders equal-price itineraries by earlier ORD arrival, then fewer flights."""
    term = TIE_PER_FLIGHT
    if f.dest == DEST:
        minutes = (f.arr - DAY_START).total_seconds() / 60
        term += TIE_ARRIVAL_WEIGHT * minutes / 1440
    return term


def excluded_reason(f: Flight, scenario: Scenario) -> str | None:
    """The first rule that fixes x_f = 0 in this scenario: C4 same day, C5 airline, C6 one stop."""
    if f.dest == DEST and f.arr > DEADLINE:
        return "C4"
    if scenario.american_only and not f.is_american:
        return "C5"
    if scenario.one_stop and f.origin != ORIGIN and f.dest != DEST:
        return "C6"
    return None


@dataclass
class Itinerary:
    rank: int
    flights: list[Flight]
    method: str  # "LP" or "IP with cuts"

    @property
    def cost(self) -> int:
        return sum(f.price for f in self.flights)

    @property
    def stops(self) -> int:
        return len(self.flights) - 1

    @property
    def airports(self) -> list[str]:
        return [self.flights[0].origin] + [f.dest for f in self.flights]

    @property
    def dep(self) -> datetime:
        return self.flights[0].dep

    @property
    def arr(self) -> datetime:
        return self.flights[-1].arr


@dataclass
class ConstraintRow:
    name: str
    group: str
    expression: str
    lhs: float


@dataclass
class SolveResult:
    scenario: Scenario
    mct: int
    status: str
    itineraries: list[Itinerary]
    n_variables: int
    n_constraints: int
    n_flights_allowed: int
    integral: bool | None
    lp_objective: float | None
    duals: dict[str, float] = field(default_factory=dict)
    reduced_costs: dict[str, float] = field(default_factory=dict)
    constraints: list[ConstraintRow] = field(default_factory=list)
    lp_text: str = ""
    shortest_path_cost: float | None = None
    node_labels: dict[str, float] = field(default_factory=dict)
    reach_cost: dict[str, int] = field(default_factory=dict)


def build_problem(net: Network, scenario: Scenario, binary: bool = False, cuts=()):
    # C7: x, w >= 0. The upper bound 1 is implied by C1 to C3 (one unit of flow), so the LP leaves it
    # off; that keeps every chosen variable basic, which makes the duals readable as cost-to-reach labels.
    cat = pulp.LpBinary if binary else pulp.LpContinuous
    prob = pulp.LpProblem(f"MIA_ORD_{scenario.key}", pulp.LpMinimize)
    x = {f.id: pulp.LpVariable(f"x_{f.id}", lowBound=0, cat=cat) for f in net.flights}
    w = {(a.name, b.name): pulp.LpVariable(f"w_{a.name}__{b.name}", lowBound=0, cat=cat) for a, b in net.waits}

    prob += pulp.lpSum((f.price + tie_term(f)) * x[f.id] for f in net.flights), "total_cost"
    prob += pulp.lpSum(x[f.id] for f in net.flights if net.tail[f.id] is None) == 1, "C1_start_MIA"
    prob += pulp.lpSum(x[f.id] for f in net.flights if net.head[f.id] is None) == 1, "C2_end_ORD"

    inflow = {n: [] for n in net.nodes}
    outflow = {n: [] for n in net.nodes}
    for f in net.flights:
        if net.head[f.id] is not None:
            inflow[net.head[f.id]].append(x[f.id])
        if net.tail[f.id] is not None:
            outflow[net.tail[f.id]].append(x[f.id])
    for a, b in net.waits:
        outflow[a].append(w[(a.name, b.name)])
        inflow[b].append(w[(a.name, b.name)])
    for n in net.nodes:
        prob += pulp.lpSum(inflow[n]) - pulp.lpSum(outflow[n]) == 0, f"C3_bal_{n.name}"

    for f in net.flights:
        reason = excluded_reason(f, scenario)
        if reason:
            prob += x[f.id] == 0, f"{reason}_{GROUP_LABEL[reason]}_{f.id}"
    for i, path in enumerate(cuts, 1):
        prob += pulp.lpSum(x[f.id] for f in path) <= len(path) - 1, f"CUT{i}"
    return prob, x, w


def _solve(prob: pulp.LpProblem) -> str:
    prob.solve(pulp.PULP_CBC_CMD(msg=False))
    return pulp.LpStatus[prob.status]


def _chosen(net: Network, x) -> list[Flight]:
    return sorted((f for f in net.flights if (x[f.id].value() or 0) > 0.5), key=lambda f: f.dep)


def _lp_text(prob: pulp.LpProblem) -> str:
    with tempfile.TemporaryDirectory() as d:
        path = Path(d) / "model.lp"
        prob.writeLP(str(path))
        return path.read_text()


def _constraint_rows(prob: pulp.LpProblem) -> list[ConstraintRow]:
    rows = []
    for name, c in prob.constraints.items():
        group = "CUT" if name.startswith("CUT") else name.split("_")[0]
        lhs = sum(coef * (var.value() or 0) for var, coef in c.items())
        rows.append(ConstraintRow(name, group, str(c), lhs))
    return rows


def _graph(net: Network, scenario: Scenario) -> nx.DiGraph:
    g = nx.DiGraph()
    for f in net.flights:
        if excluded_reason(f, scenario):
            continue
        u = net.tail[f.id].name if net.tail[f.id] else ORIGIN
        v = net.head[f.id].name if net.head[f.id] else DEST
        weight = f.price + tie_term(f)
        if not g.has_edge(u, v) or g[u][v]["weight"] > weight:
            g.add_edge(u, v, weight=weight)
    for a, b in net.waits:
        g.add_edge(a.name, b.name, weight=0.0)
    return g


def _shortest_paths(net: Network, scenario: Scenario) -> tuple[float | None, dict[str, float], dict[str, int]]:
    g = _graph(net, scenario)
    if ORIGIN not in g:
        return None, {}, {}
    labels = nx.single_source_dijkstra_path_length(g, ORIGIN, weight="weight")
    reach = {}
    for n in net.nodes:
        if n.name in labels:
            cost = int(labels[n.name])  # tie term is below $1, so int() recovers whole dollars
            reach[n.airport] = min(reach.get(n.airport, cost), cost)
    return labels.get(DEST), labels, reach


def solve(net: Network, scenario: Scenario, k: int = 5) -> SolveResult:
    prob, x, w = build_problem(net, scenario)
    status = _solve(prob)
    sp_cost, labels, reach = _shortest_paths(net, scenario)
    allowed = [f for f in net.flights if excluded_reason(f, scenario) is None]
    base = dict(scenario=scenario, mct=net.mct, n_variables=len(prob.variables()), n_constraints=len(prob.constraints),
                n_flights_allowed=len(allowed), lp_text=_lp_text(prob), shortest_path_cost=sp_cost,
                node_labels=labels, reach_cost=reach)
    if status != "Optimal":
        return SolveResult(status="infeasible", itineraries=[], integral=None, lp_objective=None,
                           constraints=_constraint_rows(prob), **base)

    integral = all(abs((v.value() or 0) - round(v.value() or 0)) < 1e-6 for v in prob.variables())
    first = Itinerary(1, _chosen(net, x), "LP")
    chosen_ids = {f.id for f in first.flights}
    result = SolveResult(
        status="optimal", itineraries=[first], integral=integral, lp_objective=pulp.value(prob.objective),
        duals={name: c.pi for name, c in prob.constraints.items()},
        reduced_costs={f.id: x[f.id].dj for f in allowed if f.id not in chosen_ids},
        constraints=_constraint_rows(prob), **base,
    )
    for rank in range(2, k + 1):
        cuts = [it.flights for it in result.itineraries]
        ip, x2, _ = build_problem(net, scenario, binary=True, cuts=cuts)
        if _solve(ip) != "Optimal":
            break
        result.itineraries.append(Itinerary(rank, _chosen(net, x2), "IP with cuts"))
    for i, it in enumerate(result.itineraries[:-1], 1):
        expr = " + ".join(f"x_{f.id}" for f in it.flights) + f" <= {len(it.flights) - 1}"
        result.constraints.append(ConstraintRow(f"CUT{i}", "CUT", expr, float(len(it.flights))))
    return result


def dual_labels(result: SolveResult) -> dict[str, float]:
    """Cost-to-reach implied by the LP duals: pi_C1 + pi_node. Equals the shortest-path label on the optimal path."""
    start = result.duals.get("C1_start_MIA", 0.0)
    return {name.removeprefix("C3_bal_"): start + pi for name, pi in result.duals.items() if name.startswith("C3_bal_")}
```

def load_flights(run_dir: Path) -> list[Flight]:
    import snapshot
    from network import flight_from_row
    return [flight_from_row(r) for r in snapshot.read_rows(run_dir / "flights.csv")]


def main() -> None:
    import argparse
    import snapshot
    from network import build_network

    parser = argparse.ArgumentParser(description="Solve all four scenarios on the latest scrape and print a report.")
    parser.add_argument("--k", type=int, default=5)
    parser.add_argument("--mct", type=int, default=45)
    parser.add_argument("--partial", action="store_true", help="allow an unfinished run (for development)")
    args = parser.parse_args()
    run_dir = snapshot.latest_run() or (snapshot.latest_run(complete=False) if args.partial else None)
    if run_dir is None:
        raise SystemExit("No complete scrape found. Run `python fetch.py` first (or pass --partial).")
    net = build_network(load_flights(run_dir), args.mct)
    out = Path(__file__).parent / "out"
    out.mkdir(exist_ok=True)
    print(f"Run {run_dir.name}: {len(net.flights)} flights, {len(net.nodes)} nodes, {len(net.waits)} wait arcs, MCT {args.mct} min")
    for s in SCENARIOS:

        (out / f"{s.key}.lp").write_text(r.lp_text)

        if r.status != "optimal":
            print("  No feasible itinerary")
            continue
        check = "match" if abs(r.lp_objective - r.shortest_path_cost) < 1e-6 else "MISMATCH"
        print(f"  integrality check: {'pass' if r.integral else 'FAIL'}; networkx cross-check: {check}")
        for it in r.itineraries:
            legs = ", ".join(f"{f.flight_number} {f.origin}-{f.dest} {f.dep:%H:%M}-{f.arr:%H:%M} ${f.price}" for f in it.flights)
            print(f"  #{it.rank} ${it.cost} ({it.method}): {legs}")


if __name__ == "__main__":
    main()
